  for manifest_case in corpus.cases:

      output_path = case_dir / "output.md"
      output_bytes = output_path.read_bytes()
      try:
          annotation = parse_outline_level_output(
              output_path=output_path,
              output_bytes=output_bytes,
              input_bytes=input_bytes,
          )

          continue
      parsed_count += 1
      exact_suffix_count += output_bytes.endswith(input_bytes)

      exact_shape = exact_shape and all(
          tuple(shape[name]) == ("level", "first_heading", "line")
          for name in ("section", "chapter", "subchapter")
      )
      exact_shape_count += exact_shape
      chapter_counts[annotation.chapter.level] += 1
